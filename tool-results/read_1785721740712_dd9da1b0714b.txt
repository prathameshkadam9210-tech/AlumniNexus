    71→function formatSalary(min?: number | null, max?: number | null, currency?: string): string {
    72→  if (!min && !max) return 'Not disclosed'
    73→  const c = currency || 'INR'
    74→  const fmt = (n: number) => `${(n / 100000).toFixed(n >= 100000 ? 1 : 0)}L`
    75→  if (min && max) return `${c} ${fmt(min)} - ${fmt(max)}`
    76→  if (min) return `From ${c} ${fmt(min)}`
    77→  return `Up to ${c} ${fmt(max!)}`
    78→}

    80→function formatDate(dateStr: string): string {
    81→  return new Date(dateStr).toLocaleDateString('en-IN', {
    82→    year: 'numeric',
    83→    month: 'short',
    84→    day: 'numeric',
    85→  })
    86→}

    88→function statusColor(status: string) {
    89→  switch (status.toUpperCase()) {
    90→    case 'ACCEPTED':
    91→    case 'COMPLETED':
    92→    case 'ACTIVE':
    93→      return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-400'
    94→    case 'PENDING':
    95→    case 'REVIEWED':
    96→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-400'
    97→    case 'REJECTED':
    98→    case 'CLOSED':
    99→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-400'
   100→    default:
   101→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   102→  }

   285→function JobDetail() {
   286→  const { jobs, applications, currentUser, selectedJobId, setCurrentView, setSelectedJobId, addApplication, addToast, setLoading } = useAppStore()
   287→  const [showApplyDialog, setShowApplyDialog] = useState(false)
   288→  const [coverLetter, setCoverLetter] = useState('')
   289→  const [applying, setApplying] = useState(false)
   290→
   291→  const job = jobs.find((j) => j.id === selectedJobId)
   292→  const hasApplied = job
   293→    ? applications.some((a) => a.jobId === job.id && a.userId === currentUser?.id)
   294→    : false
   295→
   296→  const handleApply = async () => {
   297→    if (!job || !currentUser) return
   298→    if (!coverLetter.trim()) {
   299→      addToast('Please write a cover letter to apply.', 'error')
   300→      return
   301→    }
   302→    setApplying(true)
   303→    setLoading(true)
   304→    try {
   305→      const res = await fetch('/api/applications', {
   306→        method: 'POST',
   307→        headers: { 'Content-Type': 'application/json' },
   308→        body: JSON.stringify({
   309→          userId: currentUser.id,
   310→          jobId: job.id,
   311→          coverLetter: coverLetter.trim(),
   312→        }),
   313→      })
   314→      if (!res.ok) {
   315→        const err = await res.json()
   316→        addToast(err.error || 'Failed to submit application.', 'error')
   317→        return
   318→      }
   319→      const newApp = await res.json()
   320→      addApplication(newApp)
   321→      setShowApplyDialog(false)
   322→      setCoverLetter('')
   323→      addToast('Application submitted successfully!', 'success')
   324→    } catch {
   325→      addToast('Something went wrong. Please try again.', 'error')
   326→    } finally {
   327→      setApplying(false)
   328→      setLoading(false)
   329→    }
   330→  }

   332→  if (!job) {

   348→      {/* Back button */}
   349→      <Button
   350→        variant="ghost"
   351→        className="gap-2 -ml-2"
   352→        onClick={() => {
   353→          setSelectedJobId(null)
   354→          setCurrentView('jobs')
   355→        }}
   356→      >
   357→        <ArrowLeft className="h-4 w-4" />
   358→        Back to Jobs
   359→      </Button>
   360→
   361→      <Card>
   362→        <CardContent className="p-6 space-y-6">
   363→          {/* Title Row */}
   364→          <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   365→            <div>
   366→              <div className="flex items-center gap-2 flex-wrap">
   367→                <h1 className="text-2xl font-bold tracking-tight">{job.title}</h1>
   368→                <Badge variant={typeBadgeVariant(job.type)}>{job.type}</Badge>
   369→                <Badge className={statusColor(job.status)}>{job.status}</Badge>

   487→function PostJobForm({ editJob }: { editJob?: Job | null }) {
   488→  const { currentUser, setCurrentView, addJob, addToast, setLoading, jobs } = useAppStore()
   489→
   490→  const [form, setForm] = useState({
   491→    title: '',
   492→    company: '',
   493→    location: '',
   494→    type: '',
   495→    salaryMin: '',
   496→    salaryMax: '',
   497→    description: '',
   498→    requirements: '',
   499→    skills: '',
   500→    experienceRequired: '',
   501→    deadline: '',
   502→  })
   503→  const [errors, setErrors] = useState<Record<string, string>>({})
   504→
   505→  useEffect(() => {
   506→    if (editJob) {
   507→      setForm({
   508→        title: editJob.title,
   509→        company: editJob.company,
   510→        location: editJob.location,
   511→        type: editJob.type,
   512→        salaryMin: editJob.salaryMin?.toString() || '',
   513→        salaryMax: editJob.salaryMax?.toString() || '',
   514→        description: editJob.description,
   515→        requirements: editJob.requirements,
   516→        skills: editJob.skills || '',
   517→        experienceRequired: editJob.experienceRequired || '',
   518→        deadline: editJob.deadline || '',
   519→      })
   520→    }
   521→  }, [editJob])
   522→
   523→  const updateField = (field: string, value: string) => {
   524→    setForm((prev) => ({ ...prev, [field]: value }))
   525→    if (errors[field]) {
   526→      setErrors((prev) => {
   527→        const next = { ...prev }
   528→        delete next[field]
   529→        return next
   530→      })
   531→    }
   532→  }
   533→
   534→  const validate = () => {
   535→    const errs: Record<string, string> = {}
   536→    if (!form.title.trim()) errs.title = 'Job title is required'
   537→    if (!form.company.trim()) errs.company = 'Company name is required'
   538→    if (!form.location) errs.location = 'Location is required'
   539→    if (!form.type) errs.type = 'Job type is required'
   540→    if (!form.description.trim()) errs.description = 'Description is required'
   541→    if (!form.requirements.trim()) errs.requirements = 'Requirements are required'
   542→    if (form.salaryMin && isNaN(Number(form.salaryMin))) errs.salaryMin = 'Must be a number'
   543→    if (form.salaryMax && isNaN(Number(form.salaryMax))) errs.salaryMax = 'Must be a number'
   544→    setErrors(errs)
   545→    return Object.keys(errs).length === 0
   546→  }
   547→
   548→  const handleSubmit = async (e: React.FormEvent) => {
   549→    e.preventDefault()
   550→    if (!validate()) return
   551→
   552→    setLoading(true)
   553→    try {
   554→      const body: Record<string, unknown> = {
   555→        title: form.title.trim(),
   556→        company: form.company.trim(),
   557→        location: form.location,
   558→        type: form.type,
   559→        salaryMin: form.salaryMin ? Number(form.salaryMin) : null,
   560→        salaryMax: form.salaryMax ? Number(form.salaryMax) : null,
   561→        description: form.description.trim(),
   562→        requirements: form.requirements.trim(),
