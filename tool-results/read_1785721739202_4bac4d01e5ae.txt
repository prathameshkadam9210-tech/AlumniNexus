    55→function formatDate(dateStr: string): string {
    56→  const date = new Date(dateStr)
    57→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
    58→}
    59→
    60→function getTimeAgo(dateStr: string): string {
    61→  const now = Date.now()
    62→  const then = new Date(dateStr).getTime()
    63→  const diffMs = now - then
    64→  const diffMin = Math.floor(diffMs / 60000)
    65→  const diffHr = Math.floor(diffMs / 3600000)
    66→  const diffDay = Math.floor(diffMs / 86400000)
    67→
    68→  if (diffMin < 1) return 'Just now'
    69→  if (diffMin < 60) return `${diffMin}m ago`
    70→  if (diffHr < 24) return `${diffHr}h ago`
    71→  if (diffDay < 7) return `${diffDay}d ago`
    72→  return formatDate(dateStr)
    73→}

    75→function EventTypeIcon({ type, className }: { type: string; className?: string }) {
    76→  const cls = className || 'h-3 w-3'
    77→  switch (type.toLowerCase()) {
    78→    case 'workshop': return <Trophy className={cls} />
    79→    case 'campus_drive': return <Briefcase className={cls} />
    80→    case 'seminar': return <Presentation className={cls} />
    81→    case 'webinar': return <Video className={cls} />
    82→    default: return <Calendar className={cls} />
    83→  }
    84→}

   235→  const [form, setForm] = useState({
   236→    title: '',
   237→    description: '',
   238→    date: '',
   239→    time: '',
   240→    location: '',
   241→    type: 'WORKSHOP',
   242→    maxAttendees: '',
   243→    isFeatured: false,
   244→  })
   245→
   246→  const handleSubmit = async () => {
   247→    if (!form.title.trim() || !form.date.trim()) {
   248→      addToast('Title and date are required', 'error')
   249→      return
   250→    }
   251→
   252→    setCreating(true)
   253→    setLoading(true)
   254→    try {
   255→      const res = await fetch('/api/events', {
   256→        method: 'POST',

   259→          ...form,
   260→          maxAttendees: form.maxAttendees ? parseInt(form.maxAttendees) : null,

   270→      addToast('Failed to create event', 'error')

   272→      setCreating(false)
   273→      setLoading(false)

   380→function EventsViewContent() {
   381→  const { events, currentUser, addEvent, addToast, setLoading } = useAppStore()
   382→  const [filter, setFilter] = useState('all')
   383→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   384→  const [registering, setRegistering] = useState<string | null>(null)
   385→
   386→  const isAdminOrPlacement = currentUser?.role === 'ADMIN' || currentUser?.role === 'PLACEMENT_CELL'
   387→  const isStudent = currentUser?.role === 'STUDENT'
   388→
   389→  const featuredEvents = useMemo(() => events.filter(e => e.isFeatured), [events])
   390→  const filteredEvents = useMemo(() => {
   391→    let filtered = events.filter(e => !e.isFeatured)
   392→    if (filter !== 'all') {
   393→      filtered = events.filter(e => e.type.toLowerCase() === filter.toLowerCase())
   394→    }
   395→    return filtered.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   396→  }, [events, filter])
   397→
   398→  const handleRegister = async (eventId: string) => {
   399→    setRegistering(eventId)
   400→    try {
   401→      const event = events.find(e => e.id === eventId)
   402→      if (!event) return
   403→      if (event.maxAttendees && event.currentAttendees >= event.maxAttendees) {
   404→        addToast('Event is full', 'error')
   405→        return
   406→      }
   407→      // Optimistic update
   408→      addEvent({ ...event, currentAttendees: event.currentAttendees + 1 })
   409→      addToast('Registered successfully!', 'success')
   410→    } catch {
   411→      addToast('Failed to register', 'error')
   412→    } finally {
   413→      setRegistering(null)
   414→    }
   415→  }
   416→
   417→  const handleEventCreated = (event: EventItem) => {
   418→    addEvent(event)
   419→  }
   420→
   421→  const eventTypes = [
   422→    { value: 'all', label: 'All' },
   423→    { value: 'workshop', label: 'Workshops' },
   424→    { value: 'campus_drive', label: 'Campus Drives' },
   425→    { value: 'seminar', label: 'Seminars' },
   426→    { value: 'webinar', label: 'Webinars' },
   427→  ]
   428→
   429→  return (
   430→    <div className="p-6 max-w-6xl mx-auto">
   431→      {/* Header */}
   432→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 mb-6">
   433→        <div>
   434→          <h1 className="text-2xl font-bold">Events</h1>
   435→          <p className="text-muted-foreground text-sm mt-1">Discover workshops, campus drives, seminars, and more</p>
   436→        </div>
   437→        {isAdminOrPlacement && (
   438→          <Button onClick={() => setCreateDialogOpen(true)}>
   439→            <Plus className="h-4 w-4 mr-1" />
   440→            Create Event
   441→          </Button>
   442→        )}
   443→      </div>
   444→
   445→      {/* Filter Tabs */}
   446→      <Tabs value={filter} onValueChange={setFilter}>
   447→        <TabsList className="mb-4">
   448→          {eventTypes.map(type => (
   449→            <TabsTrigger key={type.value} value={type.value} className="text-xs sm:text-sm">
   450→              {type.label}
   451→            </TabsTrigger>
   452→          ))}
   453→        </TabsList>
   454→
   455→        <TabsContent value={filter} className="mt-0">
   456→          {/* Featured Events */}
   457→          {filter === 'all' && featuredEvents.length > 0 && (
   458→            <FeaturedEvents events={featuredEvents} onRegister={handleRegister} canRegister={isStudent} />
   459→          )}
   460→
   461→          {/* All Events Grid */}
   462→          {filteredEvents.length > 0 ? (
   463→            <div className="grid gap-4 sm:grid-cols-2">
   464→              {filteredEvents.map(event => (
   465→                <EventCard
   466→                  key={event.id}
   467→                  event={event}
   468→                  onRegister={handleRegister}
   469→                  canRegister={isStudent}
   470→                />
   471→              ))}
   472→            </div>
   473→          ) : (
   474→            <div className="flex flex-col items-center justify-center py-16 text-center">
   475→              <Calendar className="h-16 w-16 text-muted-foreground/20 mb-4" />
   476→              <h3 className="text-lg font-semibold mb-1">No events found</h3>
   477→              <p className="text-sm text-muted-foreground max-w-md">
   478→                {filter === 'all'
   479→                  ? 'There are no upcoming events right now. Check back later or create one if you have admin access.'
   480→                  : `No ${eventTypes.find(t => t.value === filter)?.label.toLowerCase() || ''} events available at the moment.`}
   481→              </p>
   482→            </div>
   483→          )}
   484→        </TabsContent>
   485→      </Tabs>
   486→
   487→      {/* Create Event Dialog */}
   488→      {isAdminOrPlacement && (
   489→        <CreateEventDialog
   490→          open={createDialogOpen}
   491→          onOpenChange={setCreateDialogOpen}
   492→          onCreated={handleEventCreated}
   493→        />
   494→      )}
   495→    </div>
   496→  )

   546→  const filteredNotifications = useMemo(() => {
   547→    const sorted = [...notifications].sort(
   548→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   549→    )
   550→    if (filter === 'unread') {
   551→      return sorted.filter(n => !n.isRead)
   552→    }
   553→    return sorted
   554→  }, [notifications, filter])
   555→
   556→  const unreadCount = notifications.filter(n => !n.isRead).length
   557→
   558→  const handleMarkRead = async (id: string) => {
   559→    const notif = notifications.find(n => n.id === id)
   560→    if (!notif || notif.isRead) return
   561→
   562→    try {
   563→      const res = await fetch('/api/events', {
   564→        method: 'PUT',
   565→        headers: { 'Content-Type': 'application/json' },
   566→        body: JSON.stringify({ notificationId: id, isRead: true }),
   567→      })
   568→      if (res.ok) {
   569→        markNotificationRead(id)
   570→      }
   571→    } catch {
   572→      // Still mark locally
   573→      markNotificationRead(id)
   574→    }
   575→  }
   576→
   577→  const handleMarkAllRead = async () => {
   578→    setMarkingAll(true)
   579→    setLoading(true)
   580→    try {
   581→      const unread = notifications.filter(n => !n.isRead)
   582→      await Promise.allSettled(
   583→        unread.map(async (n) => {
   584→          try {
   585→            await fetch('/api/events', {
   586→              method: 'PUT',
   587→              headers: { 'Content-Type': 'application/json' },
   588→              body: JSON.stringify({ notificationId: n.id, isRead: true }),
   589→            })
   590→            markNotificationRead(n.id)
   591→          } catch {
   592→            // ignore individual failures
   593→          }
   594→        })
   595→      )
   596→      addToast('All notifications marked as read', 'success')
   597→    } finally {
   598→      setMarkingAll(false)
   599→      setLoading(false)
   600→    }
   601→  }
   602→
   603→  return (
   604→    <div className="p-6 max-w-3xl mx-auto">
   605→      {/* Header */}
   606→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 mb-6">
   607→        <div>
   608→          <h1 className="text-2xl font-bold">Notifications</h1>
   609→          <p className="text-muted-foreground text-sm mt-1">
   610→            {unreadCount > 0
   611→              ? `You have ${unreadCount} unread notification${unreadCount > 1 ? 's' : ''}`
   612→              : 'You are all caught up!'}
   613→          </p>
   614→        </div>
   615→        {unreadCount > 0 && (
   616→          <Button
   617→            variant="outline"
   618→            size="sm"
   619→            onClick={handleMarkAllRead}
   620→            disabled={markingAll}
   621→          >
   622→            {markingAll ? (
   623→              <div className="w-4 h-4 border-2 border-primary/30 border-t-primary rounded-full animate-spin" />
   624→            ) : (
   625→              <CheckCheck className="h-4 w-4 mr-1" />
   626→            )}
   627→            Mark All Read
   628→          </Button>
   629→        )}
   630→      </div>
   631→
   632→      {/* Filter Tabs */}
   633→      <Tabs value={filter} onValueChange={v => setFilter(v as 'all' | 'unread')}>
   634→        <TabsList className="mb-4">
   635→          <TabsTrigger value="all" className="text-sm">
   636→            All
   637→            {notifications.length > 0 && (
   638→              <Badge variant="secondary" className="ml-1.5 h-5 px-1.5 text-xs">{notifications.length}</Badge>
   639→            )}
   640→          </TabsTrigger>
   641→          <TabsTrigger value="unread" className="text-sm">
   642→            Unread
   643→            {unreadCount > 0 && (
   644→              <Badge variant="secondary" className="ml-1.5 h-5 px-1.5 text-xs">{unreadCount}</Badge>
   645→            )}
   646→          </TabsTrigger>
   647→        </TabsList>
   648→
   649→        <TabsContent value={filter} className="mt-0">
   650→          <ScrollArea className="max-h-[calc(100vh-16rem)]">
   651→            <div className="space-y-2">
   652→              {filteredNotifications.length > 0 ? (
   653→                filteredNotifications.map(notif => (
   654→                  <NotificationCard key={notif.id} notification={notif} onClick={handleMarkRead} />
   655→                ))
   656→              ) : (
   657→                <div className="flex flex-col items-center justify-center py-16 text-center">
   658→                  <Bell className="h-16 w-16 text-muted-foreground/20 mb-4" />
   659→                  <h3 className="text-lg font-semibold mb-1">No notifications yet</h3>
   660→                  <p className="text-sm text-muted-foreground max-w-md">
   661→                    {filter === 'unread'
   662→                      ? 'You have no unread notifications. Everything is up to date!'
   663→                      : 'Notifications about events, applications, and updates will appear here.'}
   664→                  </p>
   665→                </div>
   666→              )}
   667→            </div>
   668→          </ScrollArea>
   669→        </TabsContent>
   670→      </Tabs>
   671→    </div>
   672→  )
   673→}
