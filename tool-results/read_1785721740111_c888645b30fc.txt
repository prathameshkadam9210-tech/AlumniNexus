    43→    43→
    44→    44→function ProfileForm({ userId }: { userId: string }) {
    45→    45→  const { currentUser, updateUser, addToast, setLoading, setCurrentView } = useAppStore()
    46→    46→  const [saving, setSaving] = useState(false)
    47→    47→  const [uploadingAvatar, setUploadingAvatar] = useState(false)
    48→    48→  const [avatarPreview, setAvatarPreview] = useState<string | null>(currentUser?.avatar || null)
    49→    49→  const [form, setForm] = useState({
    50→    50→    name: '',
    51→    51→    email: '',
    52→    52→    phone: '',
    53→    53→    bio: '',
    54→    54→    department: '',
    55→    55→    location: '',
    56→    56→    skills: '',
    57→    57→    linkedinUrl: '',
    58→    58→    batch: '',
    59→    59→    graduationYear: '',
    60→    60→    gpa: '',
    61→    61→    resumeUrl: '',
    62→    62→    company: '',
    63→    63→    designation: '',
    64→    64→    experience: '',
    65→    65→  })
    66→    66→
    67→    67→  useEffect(() => {
    68→    68→    if (currentUser) {
    69→    69→      setForm({
    70→    70→        name: currentUser.name || '',
    71→    71→        email: currentUser.email || '',
    72→    72→        phone: currentUser.phone || '',
    73→    73→        bio: currentUser.bio || '',
    74→    74→        department: currentUser.department || '',
    75→    75→        location: currentUser.location || '',
    76→    76→        skills: formatSkillsForDisplay(currentUser.skills),
    77→    77→        linkedinUrl: currentUser.linkedinUrl || '',
    78→    78→        batch: currentUser.batch || '',
    79→    79→        graduationYear: currentUser.graduationYear?.toString() || '',
    80→    80→        gpa: currentUser.gpa?.toString() || '',
    81→    81→        resumeUrl: currentUser.resumeUrl || '',
    82→    82→        company: currentUser.company || '',
    83→    83→        designation: currentUser.designation || '',
    84→    84→        experience: currentUser.experience || '',
    85→    85→      })
    86→    86→    }
    87→    87→  }, [currentUser])
    88→    88→
    89→    89→  const formatSkillsForDisplay = (skills: string | null | undefined): string => {
    90→    90→    if (!skills) return ''
    91→    91→    try {
    92→    92→      const parsed = JSON.parse(skills)
    93→    93→      if (Array.isArray(parsed)) return parsed.join(', ')
    94→    94→    } catch {
    95→    95→      // If not JSON, return as-is (comma-separated)
    96→    96→    }
    97→    97→    return skills
    98→    98→  }
    99→    99→
   100→   100→  const handleChange = (field: string, value: string) => {
   101→   101→    setForm(prev => ({ ...prev, [field]: value }))
   102→   102→  }
   103→   103→
   104→   104→  const handleSave = async () => {
   105→   105→    setSaving(true)
   106→   106→    setLoading(true)
   107→   107→    try {
   108→   108→      const skillsStr = form.skills
   109→   109→        ? JSON.stringify(form.skills.split(',').map(s => s.trim()).filter(Boolean))
   110→   110→        : null
   111→   111→
   112→   112→      const body: Record<string, unknown> = {
   113→   113→        userId,
   114→   114→        name: form.name,
   115→   115→        email: form.email,
   116→   116→        phone: form.phone || null,
   117→   117→        bio: form.bio || null,
   118→   118→        department: form.department || null,
   119→   119→        location: form.location || null,
   120→   120→        skills: skillsStr,
   121→   121→        linkedinUrl: form.linkedinUrl || null,
   122→   122→      }
   123→   123→
   124→   124→      if (currentUser?.role === 'STUDENT' || currentUser?.role === 'ALUMNI') {
   125→   125→        body.batch = form.batch || null
   126→   126→        body.graduationYear = form.graduationYear ? parseInt(form.graduationYear) : null
   127→   127→      }
   128→   128→      if (currentUser?.role === 'STUDENT') {
   129→   129→        body.gpa = form.gpa ? parseFloat(form.gpa) : null
   130→   130→        body.resumeUrl = form.resumeUrl || null
   131→   131→      }
   132→   132→      if (currentUser?.role === 'ALUMNI' || currentUser?.role === 'RECRUITER') {
   133→   133→        body.company = form.company || null
   134→   134→        body.designation = form.designation || null
   135→   135→      }
   136→   136→      if (currentUser?.role === 'ALUMNI') {
   137→   137→        body.experience = form.experience || null
   138→   138→      }
   139→   139→
   140→   140→      const res = await fetch('/api/auth', {
   141→   141→        method: 'PUT',
   142→   142→        headers: { 'Content-Type': 'application/json' },
   143→   143→        body: JSON.stringify(body),
   144→   144→      })
   145→   145→
   146→   146→      if (!res.ok) {
   147→   147→        const err = await res.json()
   148→   148→        throw new Error(err.error || 'Failed to update profile')
   149→   149→      }
   150→   150→
   151→   151→      updateUser(userId, body)
   152→   152→      addToast('Profile updated successfully', 'success')
   153→   153→    } catch (err) {
   154→   154→      addToast(err instanceof Error ? err.message : 'Failed to update profile', 'error')
   155→   155→    } finally {
   156→   156→      setSaving(false)
   157→   157→      setLoading(false)
   158→   158→    }
   159→   159→  }
   160→   160→
   161→   161→  const handleAvatarUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   162→   162→    const file = e.target.files?.[0]
   163→   163→    if (!file) return
   164→   164→    if (file.size > 2 * 1024 * 1024) {
   165→   165→      addToast('Image must be under 2MB', 'error')
   166→   166→      return
   167→   167→    }
   168→   168→    if (!['image/jpeg', 'image/png', 'image/gif', 'image/webp'].includes(file.type)) {
   169→   169→      addToast('Only JPG, PNG, GIF, WebP allowed', 'error')
   170→   170→      return
   171→   171→    }
   172→   172→
   173→   173→    setUploadingAvatar(true)
   174→   174→    try {
   175→   175→      // Show preview immediately
   176→   176→      const reader = new FileReader()
   177→   177→      reader.onload = (ev) => setAvatarPreview(ev.target?.result as string)
   178→   178→      reader.readAsDataURL(file)
   179→   179→
   180→   180→      // Upload to server
   181→   181→      const formData = new FormData()
   182→   182→      formData.append('avatar', file)
   183→   183→      formData.append('userId', userId)
   184→   184→      const res = await fetch('/api/upload/avatar', { method: 'POST', body: formData })
   185→   185→      if (!res.ok) throw new Error('Upload failed')
   186→   186→      const data = await res.json()
   187→   187→
   188→   188→      // Save avatar URL to DB
   189→   189→      await fetch('/api/auth', {
   190→   190→        method: 'PUT',
   191→   191→        headers: { 'Content-Type': 'application/json' },
   192→   192→        body: JSON.stringify({ userId, avatar: data.avatarUrl }),
   193→   193→      })
   194→   194→      updateUser(userId, { avatar: data.avatarUrl })
   195→   195→      addToast('Profile photo updated!', 'success')
   196→   196→    } catch {
   197→   197→      addToast('Failed to upload image', 'error')
   198→   198→      setAvatarPreview(currentUser?.avatar || null)
   199→   199→    } finally {
   200→   200→      setUploadingAvatar(false)
   201→   201→    }
   202→   202→  }
   203→   203→
   204→   204→  const role = currentUser?.role || 'STUDENT'
   205→   205→  const isStudent = role === 'STUDENT'
   206→   206→  const isAlumni = role === 'ALUMNI'
   207→   207→  const isRecruiter = role === 'RECRUITER'
   208→   208→
   209→   209→  return (
   210→   210→    <div className="space-y-6">
   211→   211→      {/* Avatar Section with Upload */}
   212→   212→      <div className="flex flex-col items-center gap-4">
   213→   213→        <div className="relative group cursor-pointer" onClick={() => document.getElementById('avatar-upload')?.click()}>
   214→   214→          <Avatar className="h-28 w-28 text-2xl ring-4 ring-emerald-100 dark:ring-emerald-900/50 transition-all group-hover:ring-emerald-300">
   215→   215→            {avatarPreview ? (
   216→   216→              <img src={avatarPreview} alt={currentUser?.name} className="h-full w-full object-cover rounded-full" />
   217→   217→            ) : (
   218→   218→              <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-teal-600 text-white text-3xl font-bold">
   219→   219→                {currentUser?.name
   220→   220→                  ?.split(' ')
   221→   221→                  .map(n => n[0])
   222→   222→                  .join('')
   223→   223→                  .toUpperCase()
   224→   224→                  .slice(0, 2) || 'U'}
   225→   225→              </AvatarFallback>
   226→   226→            )}
   227→   227→          </Avatar>
   228→   228→          {/* Upload overlay */}
   229→   229→          <div className="absolute inset-0 rounded-full bg-black/40 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center">
   230→   230→            {uploadingAvatar ? (
   231→   231→              <Loader2 className="h-8 w-8 text-white animate-spin" />
   232→   232→            ) : (
   233→   233→              <div className="flex flex-col items-center gap-1 text-white">
   234→   234→                <Camera className="h-6 w-6" />
   235→   235→                <span className="text-[10px] font-medium">Change</span>
   236→   236→              </div>
   237→   237→            )}
   238→   238→          </div>
   239→   239→          {currentUser?.isVerified && (
   240→   240→            <div className="absolute -bottom-1 -right-1 bg-emerald-500 text-white rounded-full p-1 z-10">
   241→   241→              <CheckCircle2 className="h-5 w-5" />
   242→   242→            </div>
   243→   243→          )}
   244→   244→          <input
   245→   245→            id="avatar-upload"
   246→   246→            type="file"
   247→   247→            accept="image/jpeg,image/png,image/gif,image/webp"
   248→   248→            className="hidden"
   249→   249→            onChange={handleAvatarUpload}
   250→   250→          />
   251→   251→        </div>
   252→   252→        <div className="flex flex-col items-center gap-2">
   253→   253→          <div className="flex items-center gap-2">
   254→   254→            <Badge variant="secondary" className="capitalize">
   255→   255→              {role.replace('_', ' ')}
   256→   256→            </Badge>
   257→   257→            {currentUser?.isVerified && (
   258→   258→              <Badge variant="outline" className="gap-1 text-emerald-600 border-emerald-300">
   259→   259→                <Shield className="h-3 w-3" />
   260→   260→                Verified
   261→   261→              </Badge>
   262→   262→            )}
   263→   263→          </div>
   264→   264→          <p className="text-xs text-muted-foreground">Click avatar to upload (max 2MB, JPG/PNG/WebP)</p>
   265→   265→        </div>
   266→   266→      </div>
   267→   267→
   268→   268→      <Separator />
   269→   269→
   270→   270→      {/* Profile Form */}
   271→   271→      <div className="grid gap-6 md:grid-cols-2">
   272→   272→        {/* Common Fields - Left Column */}
   273→   273→        <div className="space-y-4">
   274→   274→          <h3 className="text-sm font-medium text-muted-foreground uppercase tracking-wider">Personal Information</h3>
   275→   275→
   276→   276→          <div className="space-y-2">
   277→   277→            <Label htmlFor="name">Full Name</Label>
   278→   278→            <div className="relative">
   279→   279→              <UserIcon className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   280→   280→              <Input
   281→   281→                id="name"
   282→   282→                value={form.name}
   283→   283→                onChange={e => handleChange('name', e.target.value)}
   284→   284→                className="pl-9"
   285→   285→                placeholder="Your full name"
   286→   286→              />
   287→   287→            </div>
   288→   288→          </div>
   289→   289→
   290→   290→          <div className="space-y-2">
   291→   291→            <Label htmlFor="email">Email</Label>
   292→   292→            <div className="relative">
   293→   293→              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   294→   294→              <Input
   295→   295→                id="email"
   296→   296→                type="email"
   297→   297→                value={form.email}
   298→   298→                onChange={e => handleChange('email', e.target.value)}
   299→   299→                className="pl-9"
   300→   300→                placeholder="your.email@example.com"
   301→   301→              />
   302→   302→            </div>
   303→   303→          </div>

   305→   305→          <div className="space-y-2">
   306→   306→            <Label htmlFor="phone">Phone</Label>

   309→   309→              <Input
   310→   310→                id="phone"
   311→   311→                value={form.phone}
   312→   312→                onChange={e => handleChange('phone', e.target.value)}
   313→   313→                className="pl-9"
   314→   314→                placeholder="+1 234 567 8900"
   315→   315→              />
   316→   316→            </div>
   317→   317→          </div>
   318→   318→
   319→   319→          <div className="space-y-2">
   320→   320→            <Label htmlFor="bio">Bio</Label>
   321→   321→            <Textarea
   322→   322→              id="bio"
   323→   323→              value={form.bio}
   324→   324→              onChange={e => handleChange('bio', e.target.value)}
   325→   325→              placeholder="Tell us about yourself..."
   326→   326→              rows={3}
   327→   327→            />
   328→   328→          </div>
   329→   329→
   330→   330→          <div className="space-y-2">
   331→   331→            <Label htmlFor="department">Department</Label>
   332→   332→            <div className="relative">
   333→   333→              <GraduationCap className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   334→   334→              <Input
   335→   335→                id="department"
   336→   336→                value={form.department}
   337→   337→                onChange={e => handleChange('department', e.target.value)}
   338→   338→                className="pl-9"
   339→   339→                placeholder="e.g. Computer Science"
   340→   340→              />
   341→   341→            </div>
   342→   342→          </div>
   343→   343→
   344→   344→          <div className="space-y-2">
   345→   345→            <Label htmlFor="location">Location</Label>
   346→   346→            <div className="relative">
   347→   347→              <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   348→   348→              <Input
   349→   349→                id="location"
   350→   350→                value={form.location}
   351→   351→                onChange={e => handleChange('location', e.target.value)}
   352→   352→                className="pl-9"
   353→   353→                placeholder="City, Country"
   354→   354→              />
   355→   355→            </div>
   356→   356→          </div>
   357→   357→
   358→   358→          <div className="space-y-2">
   359→   359→            <Label htmlFor="skills">Skills <span className="text-muted-foreground font-normal">(comma-separated)</span></Label>
   360→   360→            <Textarea
   361→   361→              id="skills"
   362→   362→              value={form.skills}
   363→   363→              onChange={e => handleChange('skills', e.target.value)}
   364→   364→              placeholder="React, TypeScript, Node.js, Python"
   365→   365→              rows={2}
   366→   366→            />
   367→   367→          </div>
   368→   368→
   369→   369→          <div className="space-y-2">
   370→   370→            <Label htmlFor="linkedin">LinkedIn URL</Label>

   383→   383→
   384→   384→        {/* Role-Specific Fields - Right Column */}
   385→   385→        <div className="space-y-4">
   386→   386→          <h3 className="text-sm font-medium text-muted-foreground uppercase tracking-wider">
   387→   387→            {isStudent ? 'Student Details' : isAlumni ? 'Alumni Details' : 'Professional Details'}
   388→   388→          </h3>
   389→   389→
   390→   390→          {(isStudent || isAlumni) && (
   391→   391→            <>
   392→   392→              <div className="space-y-2">
   393→   393→                <Label htmlFor="batch">Batch</Label>
   394→   394→                <Input
   395→   395→                  id="batch"
   396→   396→                  value={form.batch}
   397→   397→                  onChange={e => handleChange('batch', e.target.value)}
   398→   398→                  placeholder="e.g. 2020-2024"
   399→   399→                />
   400→   400→              </div>
   401→   401→
   402→   402→              <div className="space-y-2">
   403→   403→                <Label htmlFor="graduationYear">Graduation Year</Label>
   404→   404→                <Input
   405→   405→                  id="graduationYear"
   406→   406→                  type="number"
   407→   407→                  value={form.graduationYear}
   408→   408→                  onChange={e => handleChange('graduationYear', e.target.value)}
   409→   409→                  placeholder="2024"
   410→   410→                />
   411→   411→              </div>

   414→   414→

   417→   417→              <div className="space-y-2">

   424→   424→                  onChange={e => handleChange('gpa', e.target.value)}

   427→   427→              </div>
   428→   428→

   433→   433→                  <Input
   434→   434→                    id="resumeUrl"
   435→   435→                    value={form.resumeUrl}
   436→   436→                    onChange={e => handleChange('resumeUrl', e.target.value)}

   439→   439→                  />

   442→   442→

   446→   446→                onClick={() => setCurrentView('resume-builder')}
   447→   447→              >

   449→   449→                Open Resume Builder

   453→   453→
   454→   454→          {(isAlumni || isRecruiter) && (
   455→   455→            <>
   456→   456→              <div className="space-y-2">
   457→   457→                <Label htmlFor="company">Company</Label>
   458→   458→                <div className="relative">

   460→   460→                  <Input
   461→   461→                    id="company"
   462→   462→                    value={form.company}
   463→   463→                    onChange={e => handleChange('company', e.target.value)}
   464→   464→                    className="pl-9"
   465→   465→                    placeholder="Company name"
   466→   466→                  />
   467→   467→                </div>
   468→   468→              </div>
   469→   469→
   470→   470→              <div className="space-y-2">
   471→   471→                <Label htmlFor="designation">Designation</Label>
   472→   472→                <div className="relative">
   473→   473→                  <Briefcase className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   474→   474→                  <Input
   475→   475→                    id="designation"
   476→   476→                    value={form.designation}
   477→   477→                    onChange={e => handleChange('designation', e.target.value)}
