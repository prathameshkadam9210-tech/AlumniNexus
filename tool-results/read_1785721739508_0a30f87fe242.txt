    65→    65→function renderMarkdown(text: string): React.ReactNode[] {
    66→    66→  const lines = text.split('\n')
    67→    67→  const elements: React.ReactNode[] = []
    68→    68→  let inCodeBlock = false
    69→    69→  let codeLines: string[] = []
    70→    70→  let codeKey = 0
    71→    71→
    72→    72→  for (let i = 0; i < lines.length; i++) {
    73→    73→    const line = lines[i]
    74→    74→
    75→    75→    // Code blocks
    76→    76→    if (line.startsWith('```')) {
    77→    77→      if (inCodeBlock) {
    78→    78→        elements.push(
    79→    79→          <pre key={`code-${codeKey++}`} className="bg-muted rounded-md p-3 my-2 overflow-x-auto text-sm">
    80→    80→            <code>{codeLines.join('\n')}</code>
    81→    81→          </pre>
    82→    82→        )
    83→    83→        codeLines = []
    84→    84→        inCodeBlock = false
    85→    85→      } else {
    86→    86→        inCodeBlock = true
    87→    87→        codeLines = []
    88→    88→      }
    89→    89→      continue
    90→    90→    }
    91→    91→
    92→    92→    if (inCodeBlock) {
    93→    93→      codeLines.push(line)
    94→    94→      continue
    95→    95→    }
    96→    96→
    97→    97→    // Empty lines
    98→    98→    if (line.trim() === '') {
    99→    99→      elements.push(<br key={`br-${i}`} />)
   100→   100→      continue
   101→   101→    }
   102→   102→
   103→   103→    // Headers
   104→   104→    if (line.startsWith('### ')) {
   105→   105→      elements.push(<h4 key={`h4-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(4))}</h4>)
   106→   106→      continue
   107→   107→    }
   108→   108→    if (line.startsWith('## ')) {
   109→   109→      elements.push(<h3 key={`h3-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(3))}</h3>)
   110→   110→      continue
   111→   111→    }
   112→   112→    if (line.startsWith('# ')) {
   113→   113→      elements.push(<h2 key={`h2-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(2))}</h2>)
   114→   114→      continue
   115→   115→    }
   116→   116→
   117→   117→    // Unordered list items
   118→   118→    if (line.match(/^[*\-] /)) {
   119→   119→      const content = line.replace(/^[*\-] /, '')
   120→   120→      elements.push(
   121→   121→        <div key={`li-${i}`} className="flex items-start gap-2 ml-2 my-0.5">
   122→   122→          <span className="mt-1.5 h-1.5 w-1.5 rounded-full bg-current flex-shrink-0" />
   123→   123→          <span>{renderInline(content)}</span>
   124→   124→        </div>
   125→   125→      )
   126→   126→      continue
   127→   127→    }
   128→   128→
   129→   129→    // Ordered list items
   130→   130→    if (line.match(/^\d+\. /)) {
   131→   131→      const match = line.match(/^(\d+)\. (.*)$/)
   132→   132→      if (match) {
   133→   133→        elements.push(
   134→   134→          <div key={`oli-${i}`} className="flex items-start gap-2 ml-2 my-0.5">
   135→   135→            <span className="font-medium flex-shrink-0">{match[1]}.</span>
   136→   136→            <span>{renderInline(match[2])}</span>
   137→   137→          </div>
   138→   138→        )
   139→   139→        continue
   140→   140→      }
   141→   141→    }
   142→   142→
   143→   143→    // Regular paragraph
   144→   144→    elements.push(<p key={`p-${i}`} className="my-1">{renderInline(line)}</p>)
   145→   145→  }
   146→   146→
   147→   147→  return elements
   148→   148→}
   149→   149→
   150→   150→function renderInline(text: string): React.ReactNode {
   151→   151→  // Bold: **text** or __text__
   152→   152→  const parts = text.split(/(\*\*[^*]+\*\*|__[^_]+__)/g)
   153→   153→  return parts.map((part, i) => {
   154→   154→    if (part.startsWith('**') && part.endsWith('**')) {
   155→   155→      return <strong key={i}>{part.slice(2, -2)}</strong>
   156→   156→    }
   157→   157→    if (part.startsWith('__') && part.endsWith('__')) {
   158→   158→      return <strong key={i}>{part.slice(2, -2)}</strong>
   159→   159→    }
   160→   160→    return part
   161→   161→  })
   162→   162→}

   184→   184→interface QuickActionDialogProps {
   185→   185→  open: boolean
   186→   186→  onOpenChange: (open: boolean) => void
   187→   187→  title: string
   188→   188→  description: string
   189→   189→  fields: { name: string; label: string; placeholder: string; defaultValue?: string }[]
   190→   190→  onSubmit: (values: Record<string, string>) => void
   191→   191→  loading: boolean
   192→   192→}
   193→   193→
   194→   194→function QuickActionDialog({ open, onOpenChange, title, description, fields, onSubmit, loading }: QuickActionDialogProps) {
   195→   195→  const [values, setValues] = useState<Record<string, string>>(() => {
   196→   196→    const defaults: Record<string, string> = {}
   197→   197→    fields.forEach(f => {
   198→   198→      defaults[f.name] = f.defaultValue || ''
   199→   199→    })
   200→   200→    return defaults
   201→   201→  })
   202→   202→
   203→   203→  return (
   204→   204→    <Dialog open={open} onOpenChange={onOpenChange}>
   205→   205→      <DialogContent>
   206→   206→        <DialogHeader>
   207→   207→          <DialogTitle>{title}</DialogTitle>
   208→   208→          <DialogDescription>{description}</DialogDescription>
   209→   209→        </DialogHeader>
   210→   210→        <div className="grid gap-4 py-2">
   211→   211→          {fields.map(field => (
   212→   212→            <div key={field.name} className="space-y-2">
   213→   213→              <Label htmlFor={field.name}>{field.label}</Label>
   214→   214→              {field.name === 'message' ? (
   215→   215→                <Textarea
   216→   216→                  id={field.name}
   217→   217→                  value={values[field.name] || ''}
   218→   218→                  onChange={e => setValues(prev => ({ ...prev, [field.name]: e.target.value }))}
   219→   219→                  placeholder={field.placeholder}
   220→   220→                  rows={3}
   221→   221→                />
   222→   222→              ) : (
   223→   223→                <Input
   224→   224→                  id={field.name}
   225→   225→                  value={values[field.name] || ''}
   226→   226→                  onChange={e => setValues(prev => ({ ...prev, [field.name]: e.target.value }))}
   227→   227→                  placeholder={field.placeholder}
   228→   228→                />
   229→   229→              )}
   230→   230→            </div>
   231→   231→          ))}
   232→   232→        </div>
   233→   233→        <DialogFooter>
   234→   234→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   235→   235→          <Button
   236→   236→            onClick={() => {
   237→   237→              onSubmit(values)
   238→   238→              onOpenChange(false)
   239→   239→            }}
   240→   240→            disabled={loading || Object.values(values).some(v => !v.trim())}
   241→   241→          >
   242→   242→            {loading ? (
   243→   243→              <div className="w-4 h-4 border-2 border-primary-foreground/30 border-t-primary-foreground rounded-full animate-spin" />
   244→   244→            ) : (
   245→   245→              <Sparkles className="h-4 w-4 mr-1" />
   246→   246→            )}
   247→   247→            Generate
   248→   248→          </Button>
   249→   249→        </DialogFooter>
   250→   250→      </DialogContent>
   251→   251→    </Dialog>
   252→   252→  )
   253→   253→}
