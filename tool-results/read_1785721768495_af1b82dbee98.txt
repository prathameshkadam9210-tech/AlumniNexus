   521→interface ResumeSection {
   522→  degree: string
   523→  college: string
   524→  year: string

   528→interface ResumeExperience {

   530→  role: string

   541→
   542→function ResumeBuilder({ userId }: { userId: string }) {
   543→  const { currentUser, updateUser, addToast, setLoading, setCurrentView } = useAppStore()
   544→  const [saving, setSaving] = useState(false)
   545→  const [reviewing, setReviewing] = useState(false)
   546→  const [skillInput, setSkillInput] = useState('')
   547→  const [skills, setSkills] = useState<string[]>([])
   548→  const [objective, setObjective] = useState('')
   549→  const [education, setEducation] = useState<ResumeSection>({
   550→    degree: '',
   551→    college: '',
   552→    year: '',
   553→    gpa: '',
   554→  })
   555→  const [experiences, setExperiences] = useState<ResumeExperience[]>([])
   556→  const [projects, setProjects] = useState<ResumeProject[]>([])
   557→  const [aiReview, setAiReview] = useState<{
   558→    score: number
   559→    sections: { name: string; score: number; feedback: string }[]
   560→    suggestions: string[]
   561→  } | null>(null)
   562→
   563→  // Auto-fill from profile
   564→  useEffect(() => {
   565→    if (currentUser) {
   566→      setEducation({
   567→        degree: currentUser.department || '',
   568→        college: '',
   569→        year: currentUser.graduationYear?.toString() || '',
   570→        gpa: currentUser.gpa?.toString() || '',
   571→      })
   572→      if (currentUser.skills) {
   573→        try {
   574→          const parsed = JSON.parse(currentUser.skills)
   575→          if (Array.isArray(parsed)) setSkills(parsed)
   576→        } catch {
   577→          setSkills(currentUser.skills.split(',').map(s => s.trim()).filter(Boolean))
   578→        }
   579→      }
   580→    }
   581→  }, [currentUser])
   582→
   583→  const addSkill = () => {
   584→    const trimmed = skillInput.trim()
   585→    if (trimmed && !skills.includes(trimmed)) {
   586→      setSkills(prev => [...prev, trimmed])
   587→      setSkillInput('')
   588→    }
   589→  }
   590→
   591→  const removeSkill = (skill: string) => {
   592→    setSkills(prev => prev.filter(s => s !== skill))
   593→  }
   594→
   595→  const addExperience = () => {
   596→    setExperiences(prev => [...prev, { company: '', role: '', duration: '', description: '' }])
