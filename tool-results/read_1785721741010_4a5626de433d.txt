    80→function statusColor(status: string) {
    81→  switch (status.toUpperCase()) {
    82→    case 'ACCEPTED':
    83→    case 'COMPLETED':
    84→      return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-400'
    85→    case 'PENDING':
    86→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-400'
    87→    case 'REJECTED':
    88→    case 'CANCELLED':
    89→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-400'
    90→    default:
    91→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
    92→  }
    93→}

    95→function statusIcon(status: string) {
    96→  switch (status.toUpperCase()) {
    97→    case 'ACCEPTED':
    98→    case 'COMPLETED':
    99→      return <CheckCircle2 className="h-4 w-4 text-emerald-600" />
   100→    case 'REJECTED':

   103→    default:
   104→      return <HourglassIcon className="h-4 w-4 text-amber-500" />

   108→function getInitials(name: string): string {
   109→  return name
   110→    .split(' ')
   111→    .map((n) => n[0])
   112→    .join('')
   113→    .toUpperCase()
   114→    .slice(0, 2)
