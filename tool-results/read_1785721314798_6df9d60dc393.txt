   449→interface ResumeSection {
   450→  degree: string
   451→  college: string
   452→  year: string
   453→  gpa: string
   454→}
   455→
   456→interface ResumeExperience {
   457→  company: string
   458→  role: string
   459→  duration: string
   460→  description: string
   461→}
   462→
   463→interface ResumeProject {
   464→  title: string
   465→  tech: string
   466→  description: string
   467→  link: string
   468→}
   469→
   470→function ResumeBuilder({ userId }: { userId: string }) {
   471→  const { currentUser, updateUser, addToast, setLoading, setCurrentView } = useAppStore()
   472→  const [saving, setSaving] = useState(false)
   473→  const [reviewing, setReviewing] = useState(false)
   474→  const [skillInput, setSkillInput] = useState('')
   475→  const [skills, setSkills] = useState<string[]>([])
   476→  const [objective, setObjective] = useState('')
   477→  const [education, setEducation] = useState<ResumeSection>({
   478→    degree: '',
   479→    college: '',
   480→    year: '',
   481→    gpa: '',
   482→  })
   483→  const [experiences, setExperiences] = useState<ResumeExperience[]>([])
   484→  const [projects, setProjects] = useState<ResumeProject[]>([])
   485→  const [aiReview, setAiReview] = useState<{
   486→    score: number
   487→    sections: { name: string; score: number; feedback: string }[]
   488→    suggestions: string[]
   489→  } | null>(null)
   490→
   491→  // Auto-fill from profile
   492→  useEffect(() => {
   493→    if (currentUser) {
   494→      setEducation({
   495→        degree: currentUser.department || '',
   496→        college: '',
   497→        year: currentUser.graduationYear?.toString() || '',
   498→        gpa: currentUser.gpa?.toString() || '',
   499→      })
   500→      if (currentUser.skills) {
   501→        try {
   502→          const parsed = JSON.parse(currentUser.skills)
   503→          if (Array.isArray(parsed)) setSkills(parsed)
   504→        } catch {
   505→          setSkills(currentUser.skills.split(',').map(s => s.trim()).filter(Boolean))
   506→        }
   507→      }
   508→    }
   509→  }, [currentUser])
   510→
   511→  const addSkill = () => {
   512→    const trimmed = skillInput.trim()
   513→    if (trimmed && !skills.includes(trimmed)) {
   514→      setSkills(prev => [...prev, trimmed])
   515→      setSkillInput('')
   516→    }
   517→  }
   518→
   519→  const removeSkill = (skill: string) => {
   520→    setSkills(prev => prev.filter(s => s !== skill))
   521→  }
   522→
   523→  const addExperience = () => {
   524→    setExperiences(prev => [...prev, { company: '', role: '', duration: '', description: '' }])
   525→  }
   526→
   527→  const updateExperience = (idx: number, field: keyof ResumeExperience, value: string) => {
   528→    setExperiences(prev => prev.map((exp, i) => (i === idx ? { ...exp, [field]: value } : exp)))
   529→  }
   530→
   531→  const removeExperience = (idx: number) => {
   532→    setExperiences(prev => prev.filter((_, i) => i !== idx))
   533→  }
   534→
   535→  const addProject = () => {
   536→    setProjects(prev => [...prev, { title: '', tech: '', description: '', link: '' }])
   537→  }
   538→
   539→  const updateProject = (idx: number, field: keyof ResumeProject, value: string) => {
   540→    setProjects(prev => prev.map((proj, i) => (i === idx ? { ...proj, [field]: value } : proj)))
   541→  }
   542→
   543→  const removeProject = (idx: number) => {
   544→    setProjects(prev => prev.filter((_, i) => i !== idx))
   545→  }
   546→
   547→  const handleAIReview = async () => {
   548→    setReviewing(true)
   549→    try {
   550→      const res = await fetch('/api/ai', {
   551→        method: 'POST',
   552→        headers: { 'Content-Type': 'application/json' },
   553→        body: JSON.stringify({
   554→          action: 'resume-review',
   555→          data: {
   556→            skills,
   557→            experience: experiences,
   558→            education,
   559→            projects,
   560→            objective,
   561→            targetRole: currentUser?.designation || 'Software Engineer',
   562→          },
   563→        }),
   564→      })
   565→      if (!res.ok) throw new Error('AI review failed')
   566→      const result = await res.json()
   567→      setAiReview(result)
   568→      addToast('Resume review completed', 'success')
   569→    } catch {
   570→      addToast('Failed to get AI review', 'error')
   571→    } finally {
   572→      setReviewing(false)
   573→    }
   574→  }
   575→
   576→  const handleSave = async () => {
   577→    setSaving(true)
   578→    setLoading(true)
   579→    try {
   580→      const skillsStr = skills.length > 0 ? JSON.stringify(skills) : null
   581→      const body: Record<string, unknown> = {
   582→        userId,
   583→        skills: skillsStr,
   584→        department: education.degree || null,
   585→        graduationYear: education.year ? parseInt(education.year) : null,
   586→        gpa: education.gpa ? parseFloat(education.gpa) : null,
   587→      }
   588→
   589→      const res = await fetch('/api/auth', {
   590→        method: 'PUT',
   591→        headers: { 'Content-Type': 'application/json' },
   592→        body: JSON.stringify(body),
   593→      })
   594→
   595→      if (!res.ok) throw new Error('Failed to save resume data')
   596→      updateUser(userId, body)
   597→      addToast('Resume data saved to profile', 'success')
   598→    } catch {
   599→      addToast('Failed to save resume data', 'error')
   600→    } finally {
   601→      setSaving(false)
   602→      setLoading(false)
   603→    }
   604→  }
   605→
   606→  const handleDownloadPDF = () => {
   607→    window.print()
   608→  }
   609→
   610→  const getInitials = () => {
   611→    return currentUser?.name
   612→      ?.split(' ')
   613→      .map(n => n[0])
   614→      .join('')
   615→      .toUpperCase()
