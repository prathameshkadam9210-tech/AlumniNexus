    81→// --- Utility ---
    82→function formatRelativeTime(dateStr: string): string {
    83→  const date = new Date(dateStr)
    84→  const now = new Date()
    85→  const diff = now.getTime() - date.getTime()
    86→  const mins = Math.floor(diff / 60000)
    87→  if (mins < 1) return 'just now'
    88→  if (mins < 60) return `${mins}m ago`
    89→  const hrs = Math.floor(mins / 60)
    90→  if (hrs < 24) return `${hrs}h ago`
    91→  const days = Math.floor(hrs / 24)
    92→  if (days < 7) return `${days}d ago`
    93→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })
    94→}
    95→
    96→function formatDate(dateStr: string): string {
    97→  return new Date(dateStr).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    98→}

   606→// --- Main Dashboard View ---
   607→export function DashboardView() {
   608→  const currentUser = useAppStore((s) => s.currentUser)
   609→
   610→  if (!currentUser) return null
   611→
   612→  switch (currentUser.role) {
   613→    case 'STUDENT':
   614→      return <StudentDashboard />
   615→    case 'ALUMNI':
   616→      return <AlumniDashboard />
   617→    case 'RECRUITER':
   618→      return <RecruiterDashboard />
   619→    case 'ADMIN':
   620→    case 'PLACEMENT_CELL':
   621→      return <AdminDashboard />
   622→    default:
   623→      return <StudentDashboard />
   624→  }
   625→}
