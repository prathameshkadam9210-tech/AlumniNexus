   542→function ResumeBuilder({ userId }: { userId: string }) {
   543→  const { currentUser, updateUser, addToast, setLoading, setCurrentView } = useAppStore()

   550→    degree: '',
   551→    college: '',
   552→    year: '',
   553→    gpa: '',

   559→    sections: { name: string; score: number; feedback: string }[]
   560→    suggestions: string[]
   561→  } | null>(null)
   562→
   563→  // Auto-fill from profile
   564→  useEffect(() => {
   565→    if (currentUser) {
   566→      setEducation({
   567→        degree: currentUser.department || '',
   568→        college: '',
   569→        year: currentUser.graduationYear?.toString() || '',
   570→        gpa: currentUser.gpa?.toString() || '',
   571→      })

   573→        try {
   574→          const parsed = JSON.parse(currentUser.skills)

   592→    setSkills(prev => prev.filter(s => s !== skill))

   602→
   603→  const removeExperience = (idx: number) => {
   604→    setExperiences(prev => prev.filter((_, i) => i !== idx))
   605→  }
   606→
   607→  const addProject = () => {
   608→    setProjects(prev => [...prev, { title: '', tech: '', description: '', link: '' }])
   609→  }

   612→    setProjects(prev => prev.map((proj, i) => (i === idx ? { ...proj, [field]: value } : proj)))

   615→  const removeProject = (idx: number) => {
   616→    setProjects(prev => prev.filter((_, i) => i !== idx))

   623→        method: 'POST',

   625→        body: JSON.stringify({
   626→          action: 'resume-review',
   627→          data: {
   628→            skills,
   629→            experience: experiences,
   630→            education,
   631→            projects,
   632→            objective,
   633→            targetRole: currentUser?.designation || 'Software Engineer',
   634→          },
   635→        }),
   636→      })

   640→      addToast('Resume review completed', 'success')

   642→      addToast('Failed to get AI review', 'error')

   644→      setReviewing(false)

   649→    setSaving(true)
   650→    setLoading(true)
   651→    try {
   652→      const skillsStr = skills.length > 0 ? JSON.stringify(skills) : null

   654→        userId,
   655→        skills: skillsStr,
   656→        department: education.degree || null,
   657→        graduationYear: education.year ? parseInt(education.year) : null,
   658→        gpa: education.gpa ? parseFloat(education.gpa) : null,
   659→      }
   660→
   661→      const res = await fetch('/api/auth', {
   662→        method: 'PUT',
   663→        headers: { 'Content-Type': 'application/json' },
   664→        body: JSON.stringify(body),
   665→      })

   667→      if (!res.ok) throw new Error('Failed to save resume data')
   668→      updateUser(userId, body)
   669→      addToast('Resume data saved to profile', 'success')
   670→    } catch {
   671→      addToast('Failed to save resume data', 'error')
   672→    } finally {
   673→      setSaving(false)
   674→      setLoading(false)
   675→    }
   676→  }

   679→    window.print()
   680→  }
   681→
   682→  const getInitials = () => {
   683→    return currentUser?.name
   684→      ?.split(' ')
   685→      .map(n => n[0])
   686→      .join('')
   687→      .toUpperCase()

   693→    if (score >= 60) return 'text-amber-500'

   700→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   701→        <div>
   702→          <h2 className="text-xl font-semibold">AI Resume Builder</h2>
   703→          <p className="text-sm text-muted-foreground">Build and optimize your resume with AI assistance</p>
   704→        </div>
   705→        <div className="flex gap-2 flex-wrap">
   706→          <Button variant="outline" size="sm" onClick={() => setCurrentView('profile')}>
   707→            Back to Profile
   708→          </Button>
   709→          <Button variant="outline" size="sm" onClick={handleDownloadPDF}>
   710→            <Download className="h-4 w-4 mr-1" />
   711→            Download PDF
   712→          </Button>
   713→          <Button size="sm" onClick={handleSave} disabled={saving}>
   714→            {saving ? 'Saving...' : <Save className="h-4 w-4 mr-1" />}
   715→            Save to Profile
   716→          </Button>
   717→        </div>
   718→      </div>
   719→
   720→      <div className="grid gap-6 lg:grid-cols-2">
   721→        {/* Left: Form Sections */}
   722→        <div className="space-y-4">
   723→          <Accordion type="multiple" defaultValue={['personal', 'skills', 'education', 'experience', 'projects']} className="w-full">
   724→            {/* Personal Info */}
   725→            <AccordionItem value="personal">
   726→              <AccordionTrigger>Personal Info <Badge variant="secondary" className="ml-2 text-xs">Auto-filled</Badge></AccordionTrigger>
   727→              <AccordionContent>
   728→                <div className="grid grid-cols-2 gap-3 pt-2">
   729→                  <div className="space-y-1">

   731→                    <Input value={currentUser?.name || ''} disabled />
   732→                  </div>
   733→                  <div className="space-y-1">
   734→                    <Label className="text-xs">Email</Label>
   735→                    <Input value={currentUser?.email || ''} disabled />
   736→                  </div>
   737→                  <div className="space-y-1">
   738→                    <Label className="text-xs">Phone</Label>
   739→                    <Input value={currentUser?.phone || ''} disabled />
   740→                  </div>
   741→                  <div className="space-y-1">
   742→                    <Label className="text-xs">Location</Label>
   743→                    <Input value={currentUser?.location || ''} disabled />
   744→                  </div>

   750→            <AccordionItem value="objective">

  1038→        {/* Right: Live Resume Preview */}
