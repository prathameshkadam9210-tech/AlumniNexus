    65→function renderMarkdown(text: string): React.ReactNode[] {
    66→  const lines = text.split('\n')
    67→  const elements: React.ReactNode[] = []
    68→  let inCodeBlock = false
    69→  let codeLines: string[] = []
    70→  let codeKey = 0
    71→
    72→  for (let i = 0; i < lines.length; i++) {
    73→    const line = lines[i]
    74→
    75→    // Code blocks
    76→    if (line.startsWith('```')) {
    77→      if (inCodeBlock) {
    78→        elements.push(
    79→          <pre key={`code-${codeKey++}`} className="bg-muted rounded-md p-3 my-2 overflow-x-auto text-sm">
    80→            <code>{codeLines.join('\n')}</code>
    81→          </pre>
    82→        )
    83→        codeLines = []
    84→        inCodeBlock = false
    85→      } else {
    86→        inCodeBlock = true
    87→        codeLines = []
    88→      }
    89→      continue
    90→    }
    91→
    92→    if (inCodeBlock) {
    93→      codeLines.push(line)
    94→      continue
    95→    }
    96→
    97→    // Empty lines
    98→    if (line.trim() === '') {
    99→      elements.push(<br key={`br-${i}`} />)
   100→      continue
   101→    }
   102→
   103→    // Headers
   104→    if (line.startsWith('### ')) {
   105→      elements.push(<h4 key={`h4-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(4))}</h4>)
   106→      continue
   107→    }
   108→    if (line.startsWith('## ')) {
   109→      elements.push(<h3 key={`h3-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(3))}</h3>)
   110→      continue
   111→    }
   112→    if (line.startsWith('# ')) {
   113→      elements.push(<h2 key={`h2-${i}`} className="font-semibold mt-3 mb-1">{renderInline(line.slice(2))}</h2>)
   114→      continue
   115→    }
   116→
   117→    // Unordered list items
   118→    if (line.match(/^[*\-] /)) {
   119→      const content = line.replace(/^[*\-] /, '')
   120→      elements.push(
   121→        <div key={`li-${i}`} className="flex items-start gap-2 ml-2 my-0.5">
   122→          <span className="mt-1.5 h-1.5 w-1.5 rounded-full bg-current flex-shrink-0" />
   123→          <span>{renderInline(content)}</span>
   124→        </div>
   125→      )
   126→      continue
   127→    }
   128→
   129→    // Ordered list items
   130→    if (line.match(/^\d+\. /)) {
   131→      const match = line.match(/^(\d+)\. (.*)$/)
   132→      if (match) {
   133→        elements.push(
   134→          <div key={`oli-${i}`} className="flex items-start gap-2 ml-2 my-0.5">
   135→            <span className="font-medium flex-shrink-0">{match[1]}.</span>
   136→            <span>{renderInline(match[2])}</span>
   137→          </div>
   138→        )
   139→        continue
   140→      }
   141→    }
   142→
   143→    // Regular paragraph
   144→    elements.push(<p key={`p-${i}`} className="my-1">{renderInline(line)}</p>)
   145→  }
   146→
   147→  return elements
   148→}
   149→
   150→function renderInline(text: string): React.ReactNode {
   151→  // Bold: **text** or __text__
   152→  const parts = text.split(/(\*\*[^*]+\*\*|__[^_]+__)/g)
   153→  return parts.map((part, i) => {
   154→    if (part.startsWith('**') && part.endsWith('**')) {
   155→      return <strong key={i}>{part.slice(2, -2)}</strong>
   156→    }
   157→    if (part.startsWith('__') && part.endsWith('__')) {
   158→      return <strong key={i}>{part.slice(2, -2)}</strong>
   159→    }
   160→    return part
   161→  })
   162→}

   184→interface QuickActionDialogProps {
   185→  open: boolean
   186→  onOpenChange: (open: boolean) => void
   187→  title: string
   188→  description: string
   189→  fields: { name: string; label: string; placeholder: string; defaultValue?: string }[]
   190→  onSubmit: (values: Record<string, string>) => void
   191→  loading: boolean
   192→}
   193→
   194→function QuickActionDialog({ open, onOpenChange, title, description, fields, onSubmit, loading }: QuickActionDialogProps) {
   195→  const [values, setValues] = useState<Record<string, string>>(() => {
   196→    const defaults: Record<string, string> = {}
   197→    fields.forEach(f => {
   198→      defaults[f.name] = f.defaultValue || ''
   199→    })
   200→    return defaults
   201→  })
   202→
   203→  return (
   204→    <Dialog open={open} onOpenChange={onOpenChange}>
   205→      <DialogContent>
   206→        <DialogHeader>
   207→          <DialogTitle>{title}</DialogTitle>
   208→          <DialogDescription>{description}</DialogDescription>
   209→        </DialogHeader>
   210→        <div className="grid gap-4 py-2">
   211→          {fields.map(field => (
   212→            <div key={field.name} className="space-y-2">
   213→              <Label htmlFor={field.name}>{field.label}</Label>
   214→              {field.name === 'message' ? (
   215→                <Textarea
   216→                  id={field.name}
   217→                  value={values[field.name] || ''}
   218→                  onChange={e => setValues(prev => ({ ...prev, [field.name]: e.target.value }))}
   219→                  placeholder={field.placeholder}
   220→                  rows={3}
   221→                />
   222→              ) : (
   223→                <Input
   224→                  id={field.name}
   225→                  value={values[field.name] || ''}
   226→                  onChange={e => setValues(prev => ({ ...prev, [field.name]: e.target.value }))}
   227→                  placeholder={field.placeholder}
   228→                />
   229→              )}
   230→            </div>
   231→          ))}
   232→        </div>
   233→        <DialogFooter>
   234→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   235→          <Button
   236→            onClick={() => {
   237→              onSubmit(values)
   238→              onOpenChange(false)
   239→            }}
   240→            disabled={loading || Object.values(values).some(v => !v.trim())}
   241→          >
   242→            {loading ? (
   243→              <div className="w-4 h-4 border-2 border-primary-foreground/30 border-t-primary-foreground rounded-full animate-spin" />
   244→            ) : (
   245→              <Sparkles className="h-4 w-4 mr-1" />
   246→            )}
   247→            Generate
   248→          </Button>
   249→        </DialogFooter>
   250→      </DialogContent>
   251→    </Dialog>
   252→  )
   253→}

   591→        </ScrollArea>

   594→      {/* Main Chat Area */}

   609→              <Bot className="h-5 w-5 text-primary" />

   611→              <Badge variant="secondary" className="text-xs hidden sm:inline-flex">GPT-Powered</Badge>
   612→            </div>
   613→          </div>
   614→        </header>
   615→
   616→        {/* Messages Area */}
   617→        <ScrollArea className="flex-1">
   618→          <div className="max-w-3xl mx-auto px-4 py-4 space-y-4">
   619→            {messages.map(msg => (

   641→                </div>

   697→            </div>
   698→          </div>
   699→
   700→          {/* Input Area */}
   701→          <div className="max-w-3xl mx-auto px-4 pb-4">

   712→              />

   714→                size="icon"
   715→                onClick={() => sendMessage(input)}
   716→                disabled={!input.trim() || isTyping}
   717→                className="h-8 w-8 rounded-lg shrink-0"
   718→                aria-label="Send message"
   719→              >
   720→                <Send className="h-4 w-4" />
   721→              </Button>
   722→            </div>
   723→            <p className="text-[10px] text-muted-foreground/60 text-center mt-1.5">
   724→              AI responses are generated suggestions. Verify information independently.
   725→            </p>
   726→          </div>
   727→        </div>
   728→      </div>
   729→
   730→      {/* Quick Action Dialog */}
   731→      {dialogConfig && (
