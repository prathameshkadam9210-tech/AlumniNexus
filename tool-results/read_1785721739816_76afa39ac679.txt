    91→    91→function timeAgo(dateStr: string): string {
    92→    92→  const now = new Date()
    93→    93→  const date = new Date(dateStr)
    94→    94→  const diffMs = now.getTime() - date.getTime()
    95→    95→  const diffSec = Math.floor(diffMs / 1000)
    96→    96→  const diffMin = Math.floor(diffSec / 60)
    97→    97→  const diffHr = Math.floor(diffMin / 60)
    98→    98→  const diffDay = Math.floor(diffHr / 24)
    99→    99→
   100→   100→  if (diffSec < 60) return 'Just now'
   101→   101→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`
   102→   102→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`
   103→   103→  if (diffDay === 1) return 'Yesterday'
   104→   104→  if (diffDay < 7) return `${diffDay} days ago`
   105→   105→  if (diffDay < 30) return `${Math.floor(diffDay / 7)} week${Math.floor(diffDay / 7) > 1 ? 's' : ''} ago`
   106→   106→  if (diffDay < 365) return `${Math.floor(diffDay / 30)} month${Math.floor(diffDay / 30) > 1 ? 's' : ''} ago`
   107→   107→  return `${Math.floor(diffDay / 365)} year${Math.floor(diffDay / 365) > 1 ? 's' : ''} ago`
   108→   108→}

   118→   118→function truncateHash(hash: string, chars = 8): string {
   119→   119→  if (!hash) return ''
   120→   120→  return `${hash.slice(0, chars + 2)}...${hash.slice(-chars)}`
   121→   121→}
