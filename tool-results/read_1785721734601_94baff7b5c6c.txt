    91→function timeAgo(dateStr: string): string {
    92→  const now = new Date()
    93→  const date = new Date(dateStr)
    94→  const diffMs = now.getTime() - date.getTime()
    95→  const diffSec = Math.floor(diffMs / 1000)
    96→  const diffMin = Math.floor(diffSec / 60)
    97→  const diffHr = Math.floor(diffMin / 60)
    98→  const diffDay = Math.floor(diffHr / 24)
    99→
   100→  if (diffSec < 60) return 'Just now'
   101→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`
   102→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`
   103→  if (diffDay === 1) return 'Yesterday'
   104→  if (diffDay < 7) return `${diffDay} days ago`
   105→  if (diffDay < 30) return `${Math.floor(diffDay / 7)} week${Math.floor(diffDay / 7) > 1 ? 's' : ''} ago`
   106→  if (diffDay < 365) return `${Math.floor(diffDay / 30)} month${Math.floor(diffDay / 30) > 1 ? 's' : ''} ago`
   107→  return `${Math.floor(diffDay / 365)} year${Math.floor(diffDay / 365) > 1 ? 's' : ''} ago`
   108→}

   119→  if (!hash) return ''
   120→  return `${hash.slice(0, chars + 2)}...${hash.slice(-chars)}`
   121→}

   380→                        <div className="flex items-center justify-end gap-1">
   381→                          <Button
   382→                            variant="ghost"
   383→                            size="icon"
   384→                            className="h-8 w-8"
   385→                            onClick={() => handleViewProfile(user.id)}
   386→                            title="View Profile"
   387→                          >
   388→                            <Eye className="w-4 h-4" />

   390→                          <AlertDialog open={deleteTarget === user.id} onOpenChange={(open) => !open && setDeleteTarget(null)}>

   395→                            </AlertDialogTrigger>

   402→                              </AlertDialogHeader>
   403→                              <AlertDialogFooter>
   404→                                <AlertDialogCancel>Cancel</AlertDialogCancel>
   405→                                <AlertDialogAction onClick={handleDelete} className="bg-red-600 hover:bg-red-700">
   406→                                  Delete
   407→                                </AlertDialogAction>
   408→                              </AlertDialogFooter>
   409→                            </AlertDialogContent>
   410→                          </AlertDialog>
   411→                        </div>
   412→                      </TableCell>
   413→                    </TableRow>

   415→                )}
   416→              </TableBody>
   417→            </Table>
   418→          </div>
   419→
   420→          {/* Pagination */}
   421→          {totalPages > 1 && (
   422→            <div className="flex items-center justify-between px-4 py-3 border-t">
   423→              <p className="text-sm text-muted-foreground">
   424→                Showing {(currentPage - 1) * perPage + 1}–{Math.min(currentPage * perPage, filteredUsers.length)} of {filteredUsers.length}
   425→              </p>
   426→              <div className="flex items-center gap-1">
   427→                <Button
   428→                  variant="outline"
   429→                  size="icon"
   430→                  className="h-8 w-8"
   431→                  disabled={currentPage <= 1}
   432→                  onClick={() => setCurrentPage((p) => p - 1)}
   433→                >
   434→                  <ChevronLeft className="w-4 h-4" />
   435→                </Button>
   436→                {Array.from({ length: Math.min(totalPages, 5) }, (_, i) => {
   437→                  let page: number
   438→                  if (totalPages <= 5) {
   439→                    page = i + 1
   440→                  } else if (currentPage <= 3) {
   441→                    page = i + 1
   442→                  } else if (currentPage >= totalPages - 2) {
   443→                    page = totalPages - 4 + i
   444→                  } else {
   445→                    page = currentPage - 2 + i
   446→                  }
   447→                  return (
   448→                    <Button
   449→                      key={page}
   450→                      variant={currentPage === page ? 'default' : 'outline'}
   451→                      size="icon"
   452→                      className="h-8 w-8"
   453→                      onClick={() => setCurrentPage(page)}
   454→                    >
   455→                      {page}
   456→                    </Button>
   457→                  )
   458→                })}
   459→                <Button
   460→                  variant="outline"
   461→                  size="icon"
   462→                  className="h-8 w-8"
   463→                  disabled={currentPage >= totalPages}
   464→                  onClick={() => setCurrentPage((p) => p + 1)}
   465→                >
   466→                  <ChevronRight className="w-4 h-4" />
   467→                </Button>
   468→              </div>
   469→            </div>
   470→          )}
   471→        </CardContent>
   472→      </Card>
   473→    </div>
   474→  )
   475→}

   700→          <CardHeader className="pb-3">

   704→            </CardTitle>
   705→          </CardHeader>

   707→            <div className="overflow-x-auto max-h-96 overflow-y-auto">

   712→                </div>

   715→                  <TableHeader>
   716→                    <TableRow>
   717→                      <TableHead>Month</TableHead>

   720→                      <TableHead className="text-right">Applications</TableHead>
   721→                    </TableRow>
   722→                  </TableHeader>
   723→                  <TableBody>
   724→                    {monthlyEntries.map(([month, data]) => (

   732→                  </TableBody>
   733→                </Table>
   734→              )}
   735→            </div>
   736→          </CardContent>
   737→        </Card>
   738→      </div>
   739→    </div>
   740→  )
   741→}

   745→interface PaymentItem {

   759→const PAYMENT_STATUS_COLORS: Record<string, string> = {

   761→  COMPLETED: 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400',

   772→export function PaymentsView() {
   773→  const currentUser = useAppStore((s) => s.currentUser)
   774→  const addToast = useAppStore((s) => s.addToast)
   775→
   776→  const [payments, setPayments] = useState<PaymentItem[]>([])
   777→  const [loading, setLoading] = useState(true)
   778→  const [payDialogOpen, setPayDialogOpen] = useState(false)

   780→  const [cardNumber, setCardNumber] = useState('')
   781→  const [cardName, setCardName] = useState('')

   786→  useEffect(() => {
   787→    const url = isAdmin ? '/api/payments' : `/api/payments?userId=${currentUser?.id}`

   789→      .then((r) => r.json())
   790→      .then((data) => setPayments(Array.isArray(data) ? data : []))

   795→  const totalPaid = payments.filter((p) => p.status === 'COMPLETED').reduce((s, p) => s + p.amount, 0)

   801→    setCardNumber('')
   802→    setCardName('')
   803→    setPayDialogOpen(true)
   804→  }

   806→  const handleProcessPayment = async () => {
   807→    if (!selectedPayment || !currentUser) return

   810→      const res = await fetch('/api/payments', {
   811→        method: 'POST',
   812→        headers: { 'Content-Type': 'application/json' },

   814→          userId: currentUser.id,
   815→          amount: selectedPayment.amount,
   816→          currency: selectedPayment.currency,

   818→          description: selectedPayment.description,

   842→  if (loading) {

   844→      <div className="space-y-6">
   845→        <Skeleton className="h-8 w-48" />
   846→        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   847→          <Skeleton className="h-28 rounded-lg" />
   848→          <Skeleton className="h-28 rounded-lg" />
   849→          <Skeleton className="h-28 rounded-lg" />
   850→        </div>
   851→        <Skeleton className="h-64 rounded-lg" />
   852→      </div>
   853→    )
   854→  }
   855→
   856→  return (
   857→    <div className="space-y-6">
   858→      <div>
   859→        <h2 className="text-2xl font-bold tracking-tight">Payments</h2>
   860→        <p className="text-muted-foreground text-sm">
   861→          {isAdmin ? 'View and manage all platform payments.' : 'View your payment history and manage pending payments.'}
   862→        </p>
   863→      </div>
   864→
   865→      {/* Stats Cards */}
   866→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   867→        <Card>
   868→          <CardContent className="p-4 flex items-center gap-3">
   869→            <div className="p-2 rounded-lg bg-green-100 dark:bg-green-900/30">
   870→              <DollarSign className="w-5 h-5 text-green-600" />
   871→            </div>
   872→            <div>
   873→              <p className="text-2xl font-bold">{formatCurrency(totalPaid)}</p>
   874→              <p className="text-sm text-muted-foreground">Total Paid</p>
   875→            </div>
   876→          </CardContent>
   877→        </Card>
   878→        <Card>
   879→          <CardContent className="p-4 flex items-center gap-3">
   880→            <div className="p-2 rounded-lg bg-yellow-100 dark:bg-yellow-900/30">
   881→              <Clock className="w-5 h-5 text-yellow-600" />
   882→            </div>
   883→            <div>
   884→              <p className="text-2xl font-bold">{formatCurrency(totalPending)}</p>
   885→              <p className="text-sm text-muted-foreground">Pending</p>
   886→            </div>
   887→          </CardContent>
   888→        </Card>
   889→        <Card>
   890→          <CardContent className="p-4 flex items-center gap-3">
   891→            <div className="p-2 rounded-lg bg-purple-100 dark:bg-purple-900/30">
   892→              <Heart className="w-5 h-5 text-purple-600" />
   893→            </div>
   894→            <div>

   897→            </div>
   898→          </CardContent>
   899→        </Card>
   900→      </div>
   901→
   902→      {/* Payments Table */}
   903→      <Card>

   909→                  <TableHead>Date</TableHead>
   910→                  <TableHead>Description</TableHead>
   911→                  <TableHead className="hidden sm:table-cell">Type</TableHead>
   912→                  <TableHead className="text-right">Amount</TableHead>
   913→                  <TableHead>Status</TableHead>
   914→                  <TableHead className="text-right">Actions</TableHead>
   915→                </TableRow>
   916→              </TableHeader>
   917→              <TableBody>
   918→                {payments.length === 0 ? (
   919→                  <TableRow>
   920→                    <TableCell colSpan={6} className="text-center py-12 text-muted-foreground">

   922→                      No payments found.
   923→                    </TableCell>
   924→                  </TableRow>
   925→                ) : (
   926→                  payments.map((payment) => (
   927→                    <TableRow key={payment.id}>
